  async _isGitRepository(workspaceFolder) {
    if (!workspaceFolder) return false;
    const gitDir = path.join(workspaceFolder, ".git");
    try {
      const stat = await fs.stat(gitDir);
      return stat.isDirectory();
    } catch {
      return false;
    }
  }
